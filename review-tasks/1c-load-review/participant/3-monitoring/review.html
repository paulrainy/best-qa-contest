3SET search_path = efr;
4
5-- 1. Ночью перед тестом, по расписанию, с 01:00 МСК: перестроить раздутые таблицы
6-- (после прошлого теста client_documents вырос в 2,5 раза). Закончить до 05:00.
7VACUUM FULL client_documents;
8ANALYZE client_documents;
9
10-- 2. За 10 минут до старта прогрева: свежая статистика по таблицам, которые пишет тест
11VACUUM (ANALYZE) clients;
12VACUUM (ANALYZE) accounts;
13VACUUM (ANALYZE) deposits;
14VACUUM (ANALYZE) operations;
15VACUUM (ANALYZE) tasks;
16
17-- 3. Объём критичных таблиц против прома (выгрузка из прома от 01.11.2026)
18WITH prod(table_name, prod_rows) AS (
19 VALUES ('clients', 12000000),
20 ('client_documents', 18000000),
21 ('accounts', 30000000),
22 ('deposits', 6000000),
23 ('operations', 450000000)
24),
25nt AS (
26 SELECT relname AS table_name, n_live_tup AS nt_rows
27 FROM pg_stat_user_tables
28 WHERE schemaname = 'efr'
29)
30SELECT p.table_name,
31 p.prod_rows,
32 nt.nt_rows,
33 round(nt.nt_rows::numeric / p.prod_rows, 2) AS share,
34 nt.nt_rows >= p.prod_rows * 0.7 AS ok
35FROM prod p
36JOIN nt USING (table_name)
37ORDER BY p.table_name;
